# Reports Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the read-only Reports tab: one aggregate `GET /projects/:projectId/reports/summary` endpoint and one project-scoped `/reports` screen (window, filters, four tiles, failures chart, paginated progress table).

**Architecture:** The server extends the existing reports module with a `summary` route. The filtered trend reuses the trend SQL plus one shared filter fragment; progress is one paginated rollup over `runs`/`run_items` using `LATEST_RESULT_JOIN`, plus one totals rollup over all matching runs. The web screen issues one request per view, keeps state in the URL, and reuses `DataTable`, `StatusBar`, the date helpers and chart geometry extracted from `ResultsTrend`.

**Tech Stack:** NestJS 10 + TypeScript, PostgreSQL (`pg`), class-validator; Next.js 16 + React 19; Jest (unit), Jest + supertest (e2e).

**Spec:** `docs/api-reports.md` (contract) and `docs/PRD-reports.md` (product rules). Design: `docs/DESIGN.md` §4.9. Contracts are **approved**; do not change them without Daedalus.

  ```ts
  import type { ReportSummaryQuery } from './reports.dto';

  export type ReportStatus = 'open' | 'completed';

  /**
   * Filters after the `none` sentinel is resolved, docs/api-reports.md
   * § Implementation notes. The store never casts `'none'` to uuid.
   */
  export interface ReportFilter {
    planId: string | null;
    planNone: boolean;
    milestoneId: string | null;
    milestoneNone: boolean;
    status: ReportStatus | null;
  }

  export function resolveReportFilters(query: ReportSummaryQuery): ReportFilter {
    const plan = query.plan_id;
    const milestone = query.milestone_id;
    return {
      planId: plan && plan !== 'none' ? plan : null,
      planNone: plan === 'none',
      milestoneId: milestone && milestone !== 'none' ? milestone : null,
      milestoneNone: milestone === 'none',
      status: query.status ?? null,
    };
  }

  /** The `filters` echo: raw values, so `none` survives for the client. */
  export interface ReportFiltersDto {
    plan_id: string | null;
    milestone_id: string | null;
    status: ReportStatus | null;
  }

  export function toReportFiltersDto(query: ReportSummaryQuery): ReportFiltersDto {
    return {
      plan_id: query.plan_id ?? null,
      milestone_id: query.milestone_id ?? null,
      status: query.status ?? null,
    };
  }
  ```

  ```ts
  import { StatusCounts, percentComplete } from '../runs/run-counts';
  import type { ListMeta } from '../projects/projects.service';
  import type { ProgressItemsRow, ProgressRunRow } from './reports.store';
  import type { ReportFiltersDto } from './reports.filters';

  export interface TrendBodyDto {
    buckets: TrendBucketDto[];
    totals: TrendTotalsDto;
  }

  /** The `trend` object shared by the trend route and the summary route. */
  export function toTrendBody(rows: readonly TrendRow[]): TrendBodyDto {
    const buckets: TrendBucketDto[] = rows.map((row) => {
      const statusCounts = countsOf(row);
      return {
        date: row.day,
        status_counts: statusCounts,
        total: totalRecorded(statusCounts),
        pass_rate: passRate(statusCounts),
      };
    });
    const totalsCounts = sumRecordedCounts(buckets.map((bucket) => bucket.status_counts));
    return {
      buckets,
      totals: {
        status_counts: totalsCounts,
        total: totalRecorded(totalsCounts),
        pass_rate: passRate(totalsCounts),
      },
    };
  }

  export function toTrendDto(
    projectId: string,
    window: TrendWindow,
    rows: readonly TrendRow[],
  ): TrendDto {
    return { project_id: projectId, from: window.from, to: window.to, days: window.days, ...toTrendBody(rows) };
  }
  ```

  ```ts
  export interface ProgressRunDto {
    id: string;
    name: string;
    status: 'open' | 'completed';
    plan_id: string | null;
    plan_name: string | null;
    milestone_id: string | null;
    milestone_name: string | null;
    start_date: string | null;
    due_date: string | null;
    counts: StatusCounts;
    pass_rate: number | null;
    results_in_window: number;
    last_result_at: string | null;
  }

  export interface ProgressItemsDto {
    total: number;
    with_results: number;
    untested: number;
    percent_complete: number;
  }

  export interface ProgressDto {
    data: ProgressRunDto[];
    meta: ListMeta;
    items: ProgressItemsDto;
  }

  export interface ReportSummaryDto {
    project_id: string;
    from: string;
    to: string;
    days: number;
    filters: ReportFiltersDto;
    trend: TrendBodyDto;
    progress: ProgressDto;
  }

  export function toProgressRunDto(row: ProgressRunRow): ProgressRunDto {
    const counts: StatusCounts = {
      total: row.total,
      passed: row.passed,
      failed: row.failed,
      blocked: row.blocked,
      retest: row.retest,
      skipped: row.skipped,
      untested: row.untested,
      percent_complete: percentComplete(row.total, row.untested),
    };
    return {
      id: row.id,
      name: row.name,
      status: row.completed ? 'completed' : 'open',
      plan_id: row.plan_id,
      plan_name: row.plan_name,
      milestone_id: row.milestone_id,
      milestone_name: row.milestone_name,
      start_date: row.start_date,
      due_date: row.due_date,
      counts,
      pass_rate: passRate({
        passed: row.passed,
        failed: row.failed,
        blocked: row.blocked,
        retest: row.retest,
        skipped: row.skipped,
      }),
      results_in_window: row.results_in_window,
      last_result_at: row.last_result_at ? row.last_result_at.toISOString() : null,
    };
  }

  export function toProgressItemsDto(row: ProgressItemsRow): ProgressItemsDto {
    const untested = row.total - row.with_results;
    return {
      total: row.total,
      with_results: row.with_results,
      untested,
      percent_complete: percentComplete(row.total, untested),
    };
  }

  export function toReportSummaryDto(
    projectId: string,
    window: TrendWindow,
    filters: ReportFiltersDto,
    trendRows: readonly TrendRow[],
    progressRows: readonly ProgressRunRow[],
    meta: ListMeta,
    items: ProgressItemsRow,
  ): ReportSummaryDto {
    return {
      project_id: projectId,
      from: window.from,
      to: window.to,
      days: window.days,
      filters,
      trend: toTrendBody(trendRows),
      progress: {
        data: progressRows.map(toProgressRunDto),
        meta,
        items: toProgressItemsDto(items),
      },
    };
  }
  ```

  `percentComplete(total, untested)` is the only rounding authority for both `counts.percent_complete` and `items.percent_complete`; do not re-round in the page.

  ```ts
  function trendSql(filterSql: string, fromParam: string, toParam: string): string {
    return `
  WITH days AS (
    SELECT generate_series(${fromParam}::date, ${toParam}::date, interval '1 day')::date AS day
  ), agg AS (
    SELECT (res.created_at AT TIME ZONE 'UTC')::date AS day,
           count(*) FILTER (WHERE res.status = 'passed')::int  AS passed,
           count(*) FILTER (WHERE res.status = 'failed')::int  AS failed,
           count(*) FILTER (WHERE res.status = 'blocked')::int AS blocked,
           count(*) FILTER (WHERE res.status = 'retest')::int  AS retest,
           count(*) FILTER (WHERE res.status = 'skipped')::int AS skipped
      FROM results res
      JOIN run_items i ON i.id = res.run_item_id
      JOIN runs r ON r.id = i.run_id
     WHERE r.project_id = $1
       AND r.deleted_at IS NULL
       ${filterSql}
       AND res.created_at >= (${fromParam}::date::timestamp AT TIME ZONE 'UTC')
       AND res.created_at <  ((${toParam}::date + 1)::timestamp AT TIME ZONE 'UTC')
     GROUP BY 1
  )
  SELECT d.day::text AS day,
         coalesce(a.passed, 0)::int  AS passed,
         coalesce(a.failed, 0)::int  AS failed,
         coalesce(a.blocked, 0)::int AS blocked,
         coalesce(a.retest, 0)::int  AS retest,
         coalesce(a.skipped, 0)::int AS skipped
    FROM days d
    LEFT JOIN agg a USING (day)
  ```

  ```ts
  /**
   * URL-as-state for the reports screen (docs/DESIGN.md §4.9). The URL is the
   * source of truth for the window, the three filters and the progress page.
   *
   * Defaults are omitted from the canonical URL: `days` 30 and `page` 1, the
   * same convention as runs/cases. `none` is a real filter value ("no plan" /
   * "no milestone"), not an absent one, so it is preserved.
   *
   * URL keys are `?plan` and `?milestone` (PRD-reports.md, DESIGN §4.9); the
   * API query is `plan_id`/`milestone_id`, mapped at the request site.
   */
  import { RUN_STATUSES, type RunStatus } from "./api-types";
  import { TREND_WINDOWS, type TrendWindow } from "./trend-window";

  /** Reports defaults to 30 days; the overview's 14 is not reused. */
  export const DEFAULT_REPORT_DAYS: TrendWindow = 30;

  const UUID_RE =
    /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  export function parseReportDays(value: string | null | undefined): TrendWindow {
    const parsed = Number(value);
    return (TREND_WINDOWS as readonly number[]).includes(parsed)
      ? (parsed as TrendWindow)
      : DEFAULT_REPORT_DAYS;
  }

  function readFilter(raw: string | null): string | null {
    if (!raw) return null;
    return raw === "none" || UUID_RE.test(raw) ? raw : null;
  }

  function readPage(raw: string | null): number {
    const parsed = Number(raw);
    return Number.isInteger(parsed) && parsed >= 1 ? parsed : 1;
  }

  export interface ReportUrlState {
    project: string | null;
    days: TrendWindow;
    planId: string | null;
    milestoneId: string | null;
    status: RunStatus | null;
    page: number;
  }

  export function parseReportParams(params: URLSearchParams): ReportUrlState {
    const status = params.get("status");
    return {
      project: params.get("project"),
      days: parseReportDays(params.get("days")),
      planId: readFilter(params.get("plan")),
      milestoneId: readFilter(params.get("milestone")),
      status:
        status && (RUN_STATUSES as readonly string[]).includes(status)
          ? (status as RunStatus)
          : null,
      page: readPage(params.get("page")),
    };
  }

  export interface ReportUrlPatch {
    project?: string | null;
    days?: TrendWindow;
    planId?: string | null;
    milestoneId?: string | null;
    status?: RunStatus | null;
    page?: number;
  }

  function setOrDelete(params: URLSearchParams, key: string, value: string): void {
    if (value) params.set(key, value);
    else params.delete(key);
  }

  /**
   * Applies a patch and keeps the URL canonical. Any window or filter change
   * resets to page 1, because it starts a different result set.
   */
  export function patchReportParams(
    current: URLSearchParams,
    patch: ReportUrlPatch
  ): URLSearchParams {
    const params = new URLSearchParams(current.toString());

    if (patch.project !== undefined) setOrDelete(params, "project", patch.project ?? "");
    if (patch.days !== undefined) {
      setOrDelete(
        params,
        "days",
        patch.days === DEFAULT_REPORT_DAYS ? "" : String(patch.days)
      );
    }
    if (patch.planId !== undefined) setOrDelete(params, "plan", patch.planId ?? "");
    if (patch.milestoneId !== undefined) {
      setOrDelete(params, "milestone", patch.milestoneId ?? "");
    }
    if (patch.status !== undefined) setOrDelete(params, "status", patch.status ?? "");
    if (patch.page !== undefined) {
      setOrDelete(params, "page", patch.page > 1 ? String(patch.page) : "");
    }

    if (
      patch.page === undefined &&
      (patch.days !== undefined ||
        patch.planId !== undefined ||
        patch.milestoneId !== undefined ||
        patch.status !== undefined)
    ) {
      setOrDelete(params, "page", "");
    }

    return params;
  }
  ```
